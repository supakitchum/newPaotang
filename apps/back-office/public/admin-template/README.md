- `assets/css/styles.css`
- `assets/css/icons.css`
- `assets/css/*` font and referenced image files
- `assets/images/**`
- `assets/js/main.js`
- `assets/js/custom.js`
- `assets/js/defaultmenu.min.js`
- `assets/js/sticky.js`
- `assets/libs/node-waves/**`
- `assets/libs/simplebar/**`
